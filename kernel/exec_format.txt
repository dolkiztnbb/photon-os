    #define PEX_MAGIC 0x71414550 // "PE\x41\x71" в Little-Endian
    
    // 1. Идентификация и Целостность
    uint32_t magic;          // Сигнатура ("PE\x41\x71")
    uint32_t checksum;       // CRC32 / Adler32 заголовка и тела
    uint16_t header_size;    // Размер этого заголовка (для обратной совместимости)
    uint16_t flags;          // Флаги (права, разрешен ли I/O, тип приложения)

    // 2. Требования к ОС
    uint32_t min_kernel_ver; // Минимальная версия ядра (Packed: Major.Minor.Patch)

    // 3. Точки входа и Стек
    uint32_t entry_point;    // Смещение до функции main() от начала файла
    uint32_t stack_size;     // Запрашиваемый размер стека пользователя (например, 16 КБ)
    uint32_t heap_size;      // Начальный размер кучи приложения (kmalloc/malloc)

    // 4. Секции памяти (Image Layout)
    uint32_t text_offset;    // Смещение секции кода (.text) в файле
    uint32_t text_size;      // Размер секции кода
    uint32_t data_offset;    // Смещение инициализированных данных (.data)
    uint32_t data_size;      // Размер инициализированных данных
    uint32_t bss_size;       // Размер НЕинициализированных данных (.bss - зануляется загрузчиком!)

    // 5. Динамическая линковка / Релокации
    uint32_t reloc_offset;   // Смещение таблицы релокаций (если код не PIC/PIE)
    uint32_t reloc_count;    // Количество элементов релокаций
